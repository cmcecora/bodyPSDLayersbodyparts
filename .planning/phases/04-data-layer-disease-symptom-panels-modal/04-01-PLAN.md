    ```javascript
    import { readFileSync, writeFileSync, mkdirSync } from "node:fs";
    import { join, dirname } from "node:path";
    import { fileURLToPath } from "node:url";

    const __dirname = dirname(fileURLToPath(import.meta.url));
    const root = join(__dirname, "..");
    const source = JSON.parse(readFileSync(join(root, "public/data/diseases.json"), "utf-8"));
    const outDir = join(root, "public/data/diseases");
    mkdirSync(outDir, { recursive: true });

    let count = 0;
    for (const [key, entries] of Object.entries(source)) {
      const stripped = entries.map(({ name }) => ({ name }));
      writeFileSync(join(outDir, `${key}.json`), JSON.stringify(stripped), "utf-8");
      count++;
    }
    console.log(`Split ${count} body-part disease files to ${outDir}`);
    ```

## Trust Boundaries

| Boundary                             | Description                                                                                     |
| ------------------------------------ | ----------------------------------------------------------------------------------------------- |
| JSON data files -> DataService       | Disease/symptom JSON loaded from public/ directory; content inserted into DOM via Lit templates |
| User search input -> filter function | Text from input field used in string comparison against data                                    |

## STRIDE Threat Register

| Threat ID | Category          | Component                            | Disposition | Mitigation Plan                                                                                                                                                                                                               |
| --------- | ----------------- | ------------------------------------ | ----------- | ----------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| T-04-01   | Tampering         | data-service.ts fetch URL            | mitigate    | assetBase is stripped of trailing slash only; DataService constructs fixed-path URLs (`/data/diseases/${key}.json`). No user-controlled path segments beyond assetBase which is set by the host app developer, not end users. |
| T-04-02   | Tampering         | Disease/symptom data rendered in DOM | mitigate    | Lit `html` template literal auto-escapes all string interpolations. NEVER use `innerHTML` or `unsafeHTML` on disease/symptom names. All data rendered via `${item.name}` in Lit templates.                                    |
| T-04-03   | Denial of Service | DataService cache growth             | accept      | Cache is bounded by number of body parts (83 max). Total cached data well under 10 MB. Cache cleared on page reload. No unbounded growth vector.                                                                              |

</threat_model>
