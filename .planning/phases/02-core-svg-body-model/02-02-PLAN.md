**SVG rendering (CRITICAL — use `svg` tagged template literal, NOT `html`):**

```typescript
private _renderSvg() {
  const organsOpacity = this.currentView !== 'sections' ? '1' : '0';
  const organsPointerEvents = this.currentView !== 'sections' ? 'auto' : 'none';
  const sectionsOpacity = this.currentView === 'sections' ? '1' : '0';
  const sectionsPointerEvents = this.currentView === 'sections' ? 'auto' : 'none';

  return svg`
    <svg viewBox="0 0 698 1698" xmlns="http://www.w3.org/2000/svg">
      <defs>
        <filter id="blue-glow">
          <feDropShadow dx="0" dy="0" stdDeviation="4" flood-color="#42a5f5" flood-opacity="0.6"/>
        </filter>
        <filter id="green-glow">
          <feDropShadow dx="0" dy="0" stdDeviation="4" flood-color="#4caf50" flood-opacity="0.6"/>
        </filter>
      </defs>

      <!-- Body silhouette background -->
      <image id="base-body" x="0" y="0" width="698" height="1698"
        href="${this._silhouetteUrl()}" pointer-events="none"/>

      <!-- Organs layer (used by organs AND organs2 views) -->
      <g id="organs-layer" class="svg-layer"
        style="opacity: ${organsOpacity}; pointer-events: ${organsPointerEvents}"
        @click="${this._handleOrganClick}"
        @mouseover="${this._handleOrganHover}"
        @mouseout="${this._handleOrganOut}">
        ${ORGANS.map(organ => this._renderOrganGroup(organ))}
      </g>

      <!-- Sections layer (front only — back deferred to Phase 6 per D-06) -->
      <g id="sections-layer" class="svg-layer"
        style="opacity: ${sectionsOpacity}; pointer-events: ${sectionsPointerEvents}"
        @click="${this._handleSectionClick}">
        ${SECTIONS.filter(s => s.side === 'front').map(section => this._renderSectionGroup(section))}
      </g>
    </svg>
  `;
}
```

**Event handlers (event delegation per Pattern 4):**

```typescript
private _handleOrganClick(e: MouseEvent) {
  const group = (e.target as Element).closest('.body-part-group');
  if (!group) return;
  const partId = group.getAttribute('data-part');
  if (!partId) return;

  if (this.currentView === 'organs2') {
    // organs2: emit event for Phase 4 modal — no persistent selection
    this.dispatchEvent(new CustomEvent('organ2-click', {
      detail: { partId },
      bubbles: true,
      composed: true,
    }));
    return;
  }

  // organs view: toggle selection (MODEL-04)
  if (this._selectedOrgans.has(partId)) {
    this._selectedOrgans.delete(partId);
  } else {
    this._selectedOrgans.add(partId);
  }
  this.requestUpdate();
  this._emitSelectionChange();
}

private _handleOrganHover(e: MouseEvent) {
  // CSS handles the visual hover state — no JS needed
  // This handler exists for future tooltip support (Phase 3+)
}

private _handleOrganOut(e: MouseEvent) {
  // CSS handles the visual hover state removal
}

private _handleSectionClick(e: MouseEvent) {
  const group = (e.target as Element).closest('.body-section-group');
  if (!group) return;
  const partId = group.getAttribute('data-part');
  if (!partId) return;

  // Toggle section selection — use querySelectorAll for paired sections (Pitfall 7)
  if (this._selectedSections.has(partId)) {
    this._selectedSections.delete(partId);
  } else {
    this._selectedSections.add(partId);
  }

  // Apply selected class to ALL groups with this data-part (handles left/right pairs)
  const allGroups = this.shadowRoot!.querySelectorAll(`[data-part="${partId}"]`);
  allGroups.forEach(g => {
    if (this._selectedSections.has(partId)) {
      g.classList.add('selected');
    } else {
      g.classList.remove('selected');
    }
  });

  this.requestUpdate();
}
```
